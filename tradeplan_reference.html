۸. پیاده‌سازی MQL4

// ─── Ladder helpers ──────────────────────────────────────────────── int TrigMinutes(int tfMin) { if(tfMin <= 15) return 1; // M1, M5, M15 → trigger M1 if(tfMin == 60) return 5; // H1 → trigger M5 if(tfMin == 240) return 15; // H4 → trigger M15 if(tfMin == 1440) return 60; // D1 → trigger H1 if(tfMin == 10080) return 240; // W1 → trigger H4 return 1440; // MN → trigger D1 } int StrMinutes(int tfMin) { if(tfMin == 1) return 15; // M1 → M15 if(tfMin == 5) return 60; // M5 → H1 if(tfMin == 15) return 240; // M15 → H4 if(tfMin == 60) return 1440; // H1 → D1 if(tfMin == 240) return 10080; // H4 → W1 return 43200; // D1/W1/MN → MN } // ─── Eng (Session ATR) ───────────────────────────────────────────── double Eng(string sym, int tfMin) { int trig = TrigMinutes(tfMin); int N = tfMin / trig; return iATR(sym, trig, N, 1) / GetCachedPipSize(sym); } // ─── Trade Plan ──────────────────────────────────────────────────── void CalcPlan(string sym, int tfMin, double &SL, double &TP1, double &TP2, double &TP3, double &Eng_out, double &Hunter, double &SB1, double &SB2) { double engStr = Eng(sym, StrMinutes(tfMin)); double slRaw = 1.20 * engStr; SL = MathRound(slRaw); TP1 = MathRound(slRaw * 7.0 / 3.0); TP2 = MathRound(slRaw * 5.0); TP3 = MathRound(slRaw * 31.0 / 3.0); Eng_out = MathRound(Eng(sym, tfMin)); Hunter = MathRound(Eng_out * 8.0 / 3.0); double base = MathRound(SL * 95.0 / 9.0); double width = MathRound(SL * 20.0 / 9.0); if(tfMin == 43200) { SB1 = base + width; SB2 = base; } // MN else if(tfMin == 10080) { SB1 = MathRound(SL*16.0/3.0); SB2 = base; } // W1 else { SB1 = width; SB2 = base; } // M1-D1 }

۹. نکات کلیدی

  • Symbol-agnostic: هیچ عدد hardcoded مخصوص طلا/فارکس نیست — همه از GetCachedPipSize()
  • slRaw: تارگت‌ها از 1.20 × engStr قبل از round — نه از SL گردشده (TP3 دو واحد خطا)
  • shift=1: همه iATR ها shift=1 — کندل بسته، پایدار
  • Dynamic denominator: W1/MN بدون هیستوری کافی → وزن‌های آن پریودها از مخرج حذف
  • MN SB1: جمع ارقام گردشده: 12451 + 2621 = 15072 (نه round جمع = 15073)
  • فریز: SL/TP/SB هر bar یک‌بار — Eng و Hunter زنده (هر تیک)
